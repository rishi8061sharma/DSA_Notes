Session 6 : Aggregates and Group by working session


– Project Workload Analysis by Country


select p.project_name, e.country, count(distinct e.emp_id) as employee_count
from project p
join assignment a
on p.project_id = a.project_id
join employee e
on e.emp_id = a.emp_id
where p.project_status = 'Active'
group by p.project_name, e.country
order by p.project_name, e.country;




– Average Skill Proficiency by Department


select d.dept_name, s.skill_name, round(avg(es.proficiency_level),2) as avg_proficiency
from department d
join employee e 
on d.department_id = e.department_id
join emp_skill es
on es.emp_id = e.emp_id
join skill s
on s.skill_id = es.skill_id
where es.verified_flag = 'Y'
group by s.skill_name, d.dept_name
having count(distinct e.emp_id) >= 3
order by d.dept_name, s.skill_name;




–Overloaded Employees with Skill and Assignment Details


select e.emp_name,
   d.dept_name,
   e.country,
   SUM(a.allocation_pct) AS total_allocation,
  GROUP_CONCAT(DISTINCT s.skill_name) AS skills
from employee e
join assignment a on e.emp_id = a.emp_id
join project p on p.project_id = a.project_id and p.project_status = 'Active'
join department d on e.department_id = d.department_id
left join emp_skill es on e.emp_id = es.emp_id AND es.verified_at != '1900-01-01'
left join skill s on s.skill_id = es.skill_id 
group by e.emp_id, e.emp_name, d.dept_name, e.country
having total_allocation > 100
ORDER BY total_allocation DESC;

-- Using multiple CTEs in s query:
-- To find the top 5 customers in the Sakila database based on how much total payment
-- their name,
-- number of rentals they made,
-- total amount paid, and
-- average payment amount per customer.






-- Step 1: Count total rentals for each customer
select customer_id, count(*) as total_rentals
from rental
group by customer_id;


-- Step 2: Calculate total and average payment per customer:
select customer_id, sum(amount) as total_payment, avg(amount) as avg_payment
from payment
group by customer_id;


-- create 2 CTEs
with customer_rental as (
select customer_id, count(*) as total_rentals
from rental
group by customer_id
),
customer_payments as (
select customer_id, sum(amount) as total_payment, avg(amount) as avg_payment
from payment
group by customer_id
)
select c.first_name, cr.total_rentals, cp.total_payment, cp.avg_payment
from customer c
join customer_rental cr on c.customer_id = cr.customer_id
join customer_payments cp on c.customer_id = cp.customer_id
order by cp.total_payment desc
limit 5;

-- Rank the movies by number of rentals:

select 
f.film_id,
f.title,
count(r.rental_id) as total_rental,
RANK() over(order by count(r.rental_id)) as normal_rank,
DENSE_RANK() over(order by count(r.rental_id)) as denseRank,

on i.inventory_id = r.inventory_id
